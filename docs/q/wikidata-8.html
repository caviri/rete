PREFIX wdt: <http://www.wikidata.org/prop/direct/> PREFIX wd: <http://www.wikidata.org/entity/> PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#> PREFIX schema: <http://schema.org/> SELECT ?p ?who ?year ?occupation ?description ?image WHERE { { SELECT ?p WHERE { ?p wdt:P19 wd:Q90 } LIMIT 25 } ?p rdfs:label ?who . FILTER(LANG(?who) = "en") OPTIONAL { ?p wdt:P569 ?dob . BIND(YEAR(?dob) AS ?year) } OPTIONAL { ?p wdt:P106 ?occ . ?occ rdfs:label ?occupation . FILTER(LANG(?occupation) = "en") } OPTIONAL { ?p schema:description ?description . FILTER(LANG(?description) = "en") } OPTIONAL { ?p wdt:P18 ?image } }
