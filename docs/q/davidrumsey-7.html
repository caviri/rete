The query

PREFIX dct: <http://purl.org/dc/terms/> PREFIX drm: <https://w3id.org/rete/davidrumsey#> PREFIX edm: <http://www.europeana.eu/schemas/edm/> SELECT ?img ?title ?year (GROUP_CONCAT(DISTINCT ?place; separator=" · ") AS ?matched) (GROUP_CONCAT(DISTINCT ?city; separator=" · ") AS ?cities) (GROUP_CONCAT(DISTINCT ?county; separator=" · ") AS ?counties) (GROUP_CONCAT(DISTINCT ?state; separator=" · ") AS ?states) (GROUP_CONCAT(DISTINCT ?region; separator=" · ") AS ?regions) (GROUP_CONCAT(DISTINCT ?country; separator=" · ") AS ?countries) (GROUP_CONCAT(DISTINCT ?world; separator=" · ") AS ?worlds) WHERE { ?m dct:spatial ?place ; dct:title ?title . OPTIONAL { ?m drm:city ?city } OPTIONAL { ?m drm:county ?county } OPTIONAL { ?m drm:stateProvince ?state } OPTIONAL { ?m drm:region ?region } OPTIONAL { ?m drm:country ?country } OPTIONAL { ?m drm:worldArea ?world } OPTIONAL { ?m drm:year ?year } OPTIONAL { ?m drm:avif4k ?img } FILTER(CONTAINS(STR(?place), "Japan")) } GROUP BY ?m ?img ?title ?year ORDER BY ?year LIMIT 50
