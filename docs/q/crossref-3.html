PREFIX cx: <https://w3id.org/rete/crossref#> PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#> SELECT ?notice ?retracted ?title ?date ?source WHERE { ?u cx:updateType "retraction" ; cx:updatingWork ?notice ; cx:updatedWork ?retracted . OPTIONAL { ?retracted rdfs:label ?title } OPTIONAL { ?u cx:updateDate ?date } OPTIONAL { ?u cx:updateSource ?source } } LIMIT 30
