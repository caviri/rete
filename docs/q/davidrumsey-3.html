The query

PREFIX dct: <http://purl.org/dc/terms/> PREFIX drm: <https://w3id.org/rete/davidrumsey#> PREFIX edm: <http://www.europeana.eu/schemas/edm/> SELECT ?img ?title ?year ?author ?wcm ?hcm ?man WHERE { ?m a drm:WallMap ; dct:title ?title ; drm:objWidthCm ?wcm ; drm:objHeightCm ?hcm . OPTIONAL { ?m drm:year ?year } OPTIONAL { ?m dct:creator ?author } OPTIONAL { ?m edm:isShownBy ?img } OPTIONAL { ?m drm:iiifManifest ?man } FILTER(?wcm > 200) } ORDER BY DESC(?wcm) LIMIT 40
