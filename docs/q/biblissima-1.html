PREFIX bd: <https://data.biblissima.fr/prop/direct/> PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#> SELECT ?city ?shelfmark ?iiif WHERE { ?ms bd:P196 ?iiif ; bd:P194 ?c ; bd:P195 ?shelfmark . ?c rdfs:label ?city . FILTER(LANG(?city) = "fr") FILTER(STRSTARTS(?city, "Paris") || STRSTARTS(?city, "Gen")) } ORDER BY ?city LIMIT 24
