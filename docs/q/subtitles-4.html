PREFIX su: <https://w3id.org/rete/subtitles#> SELECT ?start ?language ?text WHERE { ?c a su:Cue ; su:start ?start ; su:text ?text . FILTER(CONTAINS(LCASE(STR(?text)), "robot")) BIND(LANG(?text) AS ?language) } ORDER BY ?start ?language
