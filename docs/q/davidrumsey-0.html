The query

PREFIX geo: <http://www.w3.org/2003/01/geo/wgs84_pos#> PREFIX gsp: <http://www.opengis.net/ont/geosparql#> PREFIX dct: <http://purl.org/dc/terms/> PREFIX drm: <https://w3id.org/rete/davidrumsey#> PREFIX edm: <http://www.europeana.eu/schemas/edm/> SELECT ?wkt ?title ?year ?author ?img ?man WHERE { ?m geo:lat ?lat ; geo:long ?long ; gsp:asWKT ?wkt ; dct:title ?title . OPTIONAL { ?m drm:year ?year } OPTIONAL { ?m dct:creator ?author } OPTIONAL { ?m edm:isShownBy ?img } OPTIONAL { ?m drm:iiifManifest ?man } FILTER(?lat > 46.0 && ?lat < 46.9 && ?long > 6.0 && ?long < 7.2) } ORDER BY ?year LIMIT 200
