PREFIX su: <https://w3id.org/rete/subtitles#> SELECT ?language ?text WHERE { ?line a su:Line ; su:start ?s ; su:end ?e ; su:text ?text . FILTER(?s <= 62 && ?e >= 62) BIND(LANG(?text) AS ?language) FILTER(?language IN ("en","es","ja","ru","fa","zh-Hans")) } ORDER BY ?language
