PREFIX a: <https://albala.icolombina.es/arcas/ns#> PREFIX dct: <http://purl.org/dc/terms/> SELECT ?title ?date ?signatura WHERE { ?r a a:ArchivalRecord ; dct:title ?title ; a:signatura ?signatura . OPTIONAL { ?r dct:date ?date } FILTER(CONTAINS(LCASE(STR(?title)), "capellan")) } LIMIT 100
