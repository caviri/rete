Cross-source JOIN: the libraries' countries (live Wikidata)

SELECT DISTINCT ?library ?country WHERE {
  ?w fjo:heldBy ?lib .
  ?lib owl:sameAs ?wd ; rdfs:label ?library .
  ?wd wdt:P17 ?c . ?c rdfs:label ?country .
  FILTER(LANG(?country) = "en")
}
