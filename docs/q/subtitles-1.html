PREFIX su: <https://w3id.org/rete/subtitles#> SELECT ?language ?text WHERE { ?line a su:Line ; su:start ?s ; su:text ?text . FILTER(?s >= 23 && ?s < 24) BIND(LANG(?text) AS ?language) } ORDER BY ?language
